# One instance of every flagged class, in a SILENT file (a hook) and a LOUD one
# (a fixture), plus every false-positive control the .sh's comments name.
=== scripts/hooks/pre-push-corpus.sh
#!/usr/bin/env bash
sed -i "s/a/b/" "$1"
sed -i '' 's/a/b/' "$1"
grep -Rl foo .claude/skills
hits=$(rg -n foo)
find . -name x -printf '%p\n'
touch -d '1 hour ago' "$1"
touch -t 209901010000 "$1"
touch -t 200001010000 "$1"
[ "$(wc -l < "$1")" = 1 ] && echo one
readlink -f "$1"
date -d "2026-01-01" +%s
stat -c %s "$1"
# --- controls below: none of these may be flagged
# sed -i in a comment is not code
img_bytes=$(stat -f%z "$1" 2>/dev/null || stat -c%s "$1" 2>/dev/null || echo 0)
touch -d '+1 hour' "$1" 2>/dev/null \
    || touch -t 202001010000 "$1"
d=$(date -d "$x" +%s 2>/dev/null || date -jf '%Y' "$x" +%s)
n=$(wc -l < "$1" | tr -d '[:space:]'); [ "$n" = 1 ] && echo one
r=$(cd "$(dirname "$1")" && pwd); readlink -f "$1" || r=$(cd x && pwd)
grep -r foo skills/
podman run img sh -c "readlink -f /x"
_SNIP="readlink -f /y"
_script="echo; $_SNIP"
podman run img sh -c "$_script"
