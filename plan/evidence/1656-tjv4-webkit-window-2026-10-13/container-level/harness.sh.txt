#!/usr/bin/env bash
# 1656-tjv4 container-level live harness (evidence, not shipped). On a PRIVATE
# store: build images/webkit, then open a WebKit window exactly as the tray
# will: its own --internal network, the 1651 SNI forwarder answering as the
# project's names (--add-host), the enclave CA as the read-only system trust
# bundle, Wayland socket only. Probes every ruling arm. Operator state checked.
set -u
REPO="$(cd "$(dirname "$0")/../../.." && pwd)"
HERE="$REPO/target/work/webkit-window"; OUT="$HERE/out"; mkdir -p "$OUT"; find "$OUT" -mindepth 1 -delete
SRC_FW=localhost/tillandsias-chromium-framework:v56.9.27.2
SRC_ROUTER=localhost/tillandsias-router:v56.9.27.2
PORT=18443
OP_ENV=(env -u CONTAINERS_STORAGE_CONF -u CONTAINERS_CONF -u REGISTRY_AUTH_FILE -u XDG_CACHE_HOME "HOME=$HOME")
[ -z "${CONTAINERS_STORAGE_CONF:-}${CONTAINERS_CONF:-}" ] || { echo "ABORT: plain shell only"; exit 3; }
log() { echo "$*" >> "$OUT/summary.txt"; echo "$*"; }
op_state() {
    local c; for c in tillandsias-inference tillandsias-dev-inference tillandsias-router; do
        "${OP_ENV[@]}" podman inspect --format '{{.Name}} {{.Id}} {{.State.Status}} {{.State.StartedAt}}' "$c" 2>&1; done
    "${OP_ENV[@]}" podman network ls --format '{{.Name}}' 2>&1 | sort
    "${OP_ENV[@]}" podman images --format '{{.Repository}}:{{.Tag}} {{.ID}}' 2>&1 | sort
}
op_state > "$OUT/op-before"
find "$XDG_RUNTIME_DIR/tillandsias/browser" -printf '%p\n' 2>/dev/null | sort > "$OUT/op-browser-dir-before"
S="$(mktemp -d "$HERE/run.XXXXXX")"; R="$(mktemp -d "${XDG_RUNTIME_DIR:?}/tillandsias-webkit-win.XXXXXX")"
mkdir -p "$S/graph" "$R/run" "$R/tmp" "$S/netconf" "$S/home" "$S/router/preview-tls" "$S/ca" "$S/foreign"
printf '%s\n' '[storage]' 'driver = "overlay"' "graphroot = \"$S/graph\"" "runroot = \"$R/run\"" > "$S/storage.conf"
printf '%s\n' '[engine]' 'cgroup_manager = "cgroupfs"' 'events_logger = "file"' "events_logfile_path = \"$S/events.log\"" \
    "tmp_dir = \"$R/tmp\"" '[network]' "network_config_dir = \"$S/netconf\"" > "$S/containers.conf"
"${OP_ENV[@]}" podman save --format oci-archive -o "$S/fw.tar" "$SRC_FW" > /dev/null || exit 1
"${OP_ENV[@]}" podman save --format oci-archive -o "$S/router.tar" "$SRC_ROUTER" > /dev/null || exit 1
WAYLAND_SOCK="$XDG_RUNTIME_DIR/${WAYLAND_DISPLAY:-wayland-0}"
export CONTAINERS_STORAGE_CONF="$S/storage.conf" CONTAINERS_CONF="$S/containers.conf" REGISTRY_AUTH_FILE="$S/auth.json" HOME="$S/home"
unset XDG_CACHE_HOME
guard() { local g; g="$(podman info --format '{{.Store.GraphRoot}}' 2>/dev/null)"; [ "$g" = "$S/graph" ] || { echo "ABORT: store is '$g'"; exit 3; }; }
guard
podman load -q -i "$S/fw.tar" > /dev/null; podman load -q -i "$S/router.tar" > /dev/null; find "$S" -maxdepth 1 -name '*.tar' -delete
printf '%s\n' "FROM $SRC_ROUTER" 'COPY base.Caddyfile /etc/caddy/base.Caddyfile' > "$S/Containerfile.router"
podman build -q -f "$S/Containerfile.router" -t localhost/e2e-router:1 "$REPO/images/router" > /dev/null 2>&1

# ---- the WebKit image, from this branch ----
guard; t0=$(date +%s)
podman build -f "$REPO/images/webkit/Containerfile" -t localhost/e2e-webkit:1 "$REPO/images/webkit" > "$OUT/build-webkit.log" 2>&1
log "build webkit rc=$? seconds=$(( $(date +%s) - t0 )) bytes=$(podman image inspect localhost/e2e-webkit:1 --format '{{.Size}}' 2>/dev/null)"

# ---- CAs: the enclave CA (trusted) and a foreign CA (must be rejected) ----
mkca() { openssl req -x509 -newkey rsa:2048 -nodes -days 1 -subj "/CN=$2" -keyout "$1/ca.key" -out "$1/ca.crt" \
    -addext basicConstraints=critical,CA:TRUE -addext keyUsage=critical,keyCertSign,cRLSign 2>/dev/null; }
leaf() { # dir ca-dir name
    printf 'subjectAltName=DNS:%s\nbasicConstraints=critical,CA:FALSE\nkeyUsage=critical,digitalSignature,keyEncipherment\nextendedKeyUsage=serverAuth\n' "$3" > "$1/$3.ext"
    openssl req -new -newkey rsa:2048 -nodes -subj "/CN=$3" -keyout "$1/$3.key" -out "$1/$3.csr" 2>/dev/null
    openssl x509 -req -in "$1/$3.csr" -CA "$2/ca.crt" -CAkey "$2/ca.key" -set_serial "$RANDOM$RANDOM" -days 1 -extfile "$1/$3.ext" -out "$1/$3.crt" 2>/dev/null; }
mkca "$S/ca" e2e-enclave-ca; mkca "$S/foreign" e2e-foreign-ca
for p in demo other; do leaf "$S/ca" "$S/ca" "www.$p.localhost"; cat "$S/ca/www.$p.localhost.crt" "$S/ca/ca.crt" > "$S/router/preview-tls/$p.crt"; cp "$S/ca/www.$p.localhost.key" "$S/router/preview-tls/$p.key"; chmod 0644 "$S/router/preview-tls/$p.key"; done
leaf "$S/ca" "$S/ca" localhost
leaf "$S/foreign" "$S/foreign" localhost
# The window's WHOLE trust bundle: the enclave CA only (the tray writes this file).
cp "$S/ca/ca.crt" "$S/trust-bundle.crt"; chmod 0444 "$S/trust-bundle.crt"

# ---- enclave-like network, dev servers, router (as the 1649 harness) ----
HARD=(--cap-drop=ALL --security-opt=no-new-privileges --security-opt=label=disable --userns=keep-id --read-only --tmpfs /tmp:size=64m)
guard
podman network create e2e-enclave > /dev/null
WEBJS="$REPO/target/work/browser-open-e2e"
podman run -d --name e2e-demo-web --network e2e-enclave "${HARD[@]}" -v "$WEBJS:/e2e:ro" --entrypoint node "$SRC_FW" /e2e/web.js PREVIEW-DEMO > /dev/null
podman run -d --name e2e-other-web --network e2e-enclave "${HARD[@]}" -v "$WEBJS:/e2e:ro" --entrypoint node "$SRC_FW" /e2e/web.js PREVIEW-OTHER > /dev/null
{ for p in demo other; do
    printf 'https://www.%s.localhost:8443 {\n    tls /etc/tillandsias/preview/%s.crt /etc/tillandsias/preview/%s.key\n    reverse_proxy e2e-%s-web:3000\n}\n\n' "$p" "$p" "$p" "$p"
  done; } > "$S/router/dynamic.Caddyfile"
podman run -d --name tillandsias-router --hostname router --network e2e-enclave "${HARD[@]}" --pids-limit=64 --tmpfs /run/router:size=8m \
    -e http_proxy= -e https_proxy= -e no_proxy='*' \
    -v "$S/router/dynamic.Caddyfile:/run/router/dynamic.Caddyfile:rw" \
    --mount "type=bind,source=$S/ca/ca.crt,target=/etc/tillandsias/ca.crt,readonly=true" \
    --mount "type=bind,source=$S/router/preview-tls,target=/etc/tillandsias/preview,readonly=true" localhost/e2e-router:1 > /dev/null
sleep 3
ROUTER_IP="$(podman inspect --format '{{(index .NetworkSettings.Networks "e2e-enclave").IPAddress}}' tillandsias-router)"

# ---- the window: internal net, forwarder (8443 + preview port), add-host, CA bundle ----
podman network create --internal e2e-win > /dev/null
podman run -d --name e2e-fwd --network e2e-win:alias=router "${HARD[@]}" --pids-limit 32 \
    -e TILLANDSIAS_FWD_PROJECT=demo -e TILLANDSIAS_FWD_UPSTREAM_HOST="$ROUTER_IP" -e TILLANDSIAS_FWD_UPSTREAM_PORT=8443 \
    -e TILLANDSIAS_FWD_EXTRA_PORTS=$PORT --entrypoint node "$SRC_FW" \
    -e "$(cat "$REPO/crates/tillandsias-headless/src/browser_window_forwarder.js")" > /dev/null
podman network connect e2e-enclave e2e-fwd
FWD_IP="$(podman inspect --format '{{(index .NetworkSettings.Networks "e2e-win").IPAddress}}' e2e-fwd)"
sleep 1
log "router $ROUTER_IP forwarder-on-window-net $FWD_IP; fwd: $(podman logs e2e-fwd 2>&1 | tr '\n' ' ')"
DEVICE='{"viewport":{"width":393,"height":852},"deviceScaleFactor":3,"isMobile":true,"hasTouch":true,"userAgent":"Mozilla/5.0 (iPhone; CPU iPhone OS 17_5 like Mac OS X) AppleWebKit/605.1.15 (KHTML, like Gecko) Version/27.2 Mobile/15E148 Safari/604.1"}'
guard
podman run -d --name e2e-webkit-win --network e2e-win "${HARD[@]}" --pids-limit 512 --tmpfs /dev/shm:size=256m \
    --add-host "www.demo.localhost:$FWD_IP" --add-host "web.demo.localhost:$FWD_IP" \
    --mount "type=bind,source=$S/trust-bundle.crt,target=/etc/ssl/certs/ca-certificates.crt,readonly=true" \
    --tmpfs /run/user/browser:size=1m,mode=0700,U -v "$WAYLAND_SOCK:/run/user/browser/wayland-0:rw" \
    -e XDG_RUNTIME_DIR=/run/user/browser -e WAYLAND_DISPLAY=wayland-0 -e GDK_BACKEND=wayland -e HOME=/tmp \
    -e "TILLANDSIAS_WEBKIT_URL=https://www.demo.localhost:$PORT/" -e "TILLANDSIAS_WEBKIT_DEVICE=$DEVICE" \
    -e "TILLANDSIAS_WEBKIT_RELAY=$FWD_IP:$PORT" \
    localhost/e2e-webkit:1 > "$OUT/window-run.txt" 2>&1
log "window run rc=$?"
sleep 8
log "window state $(podman inspect --format '{{.State.Status}} exit={{.State.ExitCode}}' e2e-webkit-win)"
podman logs e2e-webkit-win > "$OUT/window.log" 2>&1
log "window says: $(grep -E 'webkit-window' "$OUT/window.log" | tr '\n' ' ')"
podman logs e2e-demo-web > "$OUT/demo-web.log" 2>&1
log "demo-web saw: $(grep -E '^GET / ' "$OUT/demo-web.log" | head -2 | tr '\n' ' ')"
log "page reports: $(grep -E '^GET /report' "$OUT/demo-web.log" | tail -1)"
log "window /etc/hosts: $(podman exec e2e-webkit-win cat /etc/hosts | grep -v '^#' | tr '\n' ';')"
log "window listeners: $(podman exec e2e-webkit-win cat /proc/net/tcp | awk 'NR>1 && $4=="0A"{print $2}' | tr '\n' ' ')"
log "window trust bundle: $(podman exec e2e-webkit-win sh -c 'grep -c BEGIN /etc/ssl/certs/ca-certificates.crt; touch /etc/ssl/certs/ca-certificates.crt 2>&1 | head -1')"

# ---- probes inside the window container ----
nl() { awk '{printf "%s\\n", $0}' "$1"; }
podman exec -i -e PROBE_PREVIEW_PORT=$PORT -e PLAYWRIGHT_BROWSERS_PATH=/ms-playwright -e HOME=/tmp \
    -e "PROBE_ENCLAVE_LEAF=$(nl "$S/ca/localhost.crt")" -e "PROBE_ENCLAVE_KEY=$(nl "$S/ca/localhost.key")" \
    -e "PROBE_FOREIGN_LEAF=$(nl "$S/foreign/localhost.crt")" -e "PROBE_FOREIGN_KEY=$(nl "$S/foreign/localhost.key")" \
    e2e-webkit-win timeout 120 node - < "$HERE/probe.js" > "$OUT/probe.txt" 2>&1
log "probe rc=$?"; sed 's/^/probe /' "$OUT/probe.txt" | tee -a "$OUT/summary.txt"
podman logs e2e-other-web > "$OUT/other-web.log" 2>&1
log "other-project upstream requests: $(grep -c '^GET ' "$OUT/other-web.log")"

# ---- cleanup: this private store only. NOT `podman system reset`: it also
# clears the netavark run dir, which is the operator's $XDG_RUNTIME_DIR one.
# Whole scratch dirs by find: an empty variable gives `find ""`, an error.
guard
podman rm -a -f -t 0 > /dev/null 2>&1
for n in $(podman network ls --format '{{.Name}}'); do [ "$n" = podman ] || podman network rm -f "$n" > /dev/null 2>&1; done
podman rmi -a -f > /dev/null 2>&1
if [ -n "$S" ] && [ -n "$R" ]; then
    for _ in 1 2 3; do
        podman unshare umount -l "$S/graph/overlay" 2>/dev/null
        podman unshare find "$S" "$R" -delete 2>/dev/null
        [ -e "$S" ] || [ -e "$R" ] || break
        sleep 2
    done
fi
export HOME="$(getent passwd "$(id -u)" | cut -d: -f6)"; unset CONTAINERS_STORAGE_CONF CONTAINERS_CONF REGISTRY_AUTH_FILE
op_state > "$OUT/op-after"
find "$XDG_RUNTIME_DIR/tillandsias/browser" -printf '%p\n' 2>/dev/null | sort > "$OUT/op-browser-dir-after"
for f in op op-browser-dir; do cmp -s "$OUT/$f-before" "$OUT/$f-after" && log "$f: unchanged" || { log "$f: CHANGED"; diff "$OUT/$f-before" "$OUT/$f-after" | tee -a "$OUT/summary.txt"; }; done
log "leftover scratch: $(ls -d "$HERE"/run.* "$XDG_RUNTIME_DIR"/tillandsias-webkit-win.* 2>/dev/null | tr '\n' ' ')"
