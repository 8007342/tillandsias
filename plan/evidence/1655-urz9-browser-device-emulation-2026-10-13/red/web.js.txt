// Stand-in project dev server: plain http on :3000, a fixed <title> marker,
// one stdout line per request (the live test reads the window's request here).
// 1655-urz9: the page reports what IT sees, [innerWidth, devicePixelRatio,
// navigator.maxTouchPoints] plus the UA, on load and on every change, as
// `GET /report?...` lines, so the measurement does not trust the CDP path.
const PAGE = (marker, host) => `<!doctype html><html><head>
<meta name="viewport" content="width=device-width, initial-scale=1">
<title>${marker}</title></head><body>${marker} host=${host}
<script>
let last = '';
function report() {
  const s = JSON.stringify([innerWidth, devicePixelRatio, navigator.maxTouchPoints]);
  if (s === last) return;
  last = s;
  fetch('/report?m=' + encodeURIComponent(s) + '&ua=' + encodeURIComponent(navigator.userAgent)).catch(() => {});
}
report();
setInterval(report, 250);
</script></body></html>
`;
require('http').createServer((req, res) => {
  console.log(`${req.method} ${req.url} host=${req.headers.host} ua=${req.headers['user-agent']}`);
  if (req.url.startsWith('/report')) { res.writeHead(204); res.end(); return; }
  res.writeHead(200, { 'content-type': 'text/html' });
  res.end(PAGE(process.argv[2], req.headers.host));
}).listen(3000, '0.0.0.0', () => console.log(`${process.argv[2]} listening 3000`));
